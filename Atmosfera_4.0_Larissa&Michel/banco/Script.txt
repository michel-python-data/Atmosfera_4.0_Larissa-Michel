-- Inicialização do Banco de Dados  
DROP DATABASE IF EXISTS atmosfera_4_0;
CREATE DATABASE atmosfera_4_0;
USE atmosfera_4_0;

-- Cria tabela dispositivos
CREATE TABLE dispositivos ( 
    id_dispositivos INT PRIMARY KEY AUTO_INCREMENT,
    nome_dispositivos VARCHAR(100) NOT NULL, 
    localizacao VARCHAR(100) NOT NULL, 
    tipo_dispositivos VARCHAR(100) NOT NULL, 
    gateway_pai_fk INT NULL,
    FOREIGN KEY (gateway_pai_fk) REFERENCES dispositivos(id_dispositivos)
); 

-- Cria tabela atuadores
CREATE TABLE atuadores ( 
    id_atuador INT PRIMARY KEY AUTO_INCREMENT,
    nome_atuador VARCHAR(100) NOT NULL, 
    status_atual BOOLEAN NOT NULL, 
    ultimo_comando DATETIME, 
    id_dispositivo_fk INT NOT NULL,
    FOREIGN KEY (id_dispositivo_fk) REFERENCES dispositivos(id_dispositivos)
); 

-- Cria tabela registros
CREATE TABLE registros ( 
    id_registro BIGINT PRIMARY KEY AUTO_INCREMENT,
    time_stamp TIMESTAMP NOT NULL, -- Horário da medição no dispositivo
    data_hora DATETIME DEFAULT NOW(), -- Horário do recebimento no servidor
    temperatura FLOAT,
    umidade FLOAT,
    pressao FLOAT,
    altitude FLOAT,
    luminosidade FLOAT,
    velocidade_vento FLOAT,
    direcao_vento INT,
    intensidade_wifi INT,
    id_dispositivo_fk INT NOT NULL,
    FOREIGN KEY (id_dispositivo_fk) REFERENCES dispositivos(id_dispositivos)
);
